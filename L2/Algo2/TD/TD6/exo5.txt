#include <stdio.h>

int strlen(char c) {
    int n = 0;
    for (int i = 0; c[i] != "\0"; i += 1) {
        n += 1;
    }
    return(n);
}

char reverse(char s) {
    n = strlen(s); // Longueur de s
    char t[n];
    for (int i = 0; s[i] != "\0" || i == n; i += 1) {
        t[i] = s[i];
    }
    for (int i = n; i > 0;i -= 1) {
        s[i] = t[i];
    }
    return(s);
}

int main() {
    char s[1000];
    scanf("%c", &s);
    printf("%c", reverse(s));
}
