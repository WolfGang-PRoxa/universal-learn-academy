#include <stdio.h>

int strlen(char c) {
    int n = 0;
    for (int i = 0; c[i] != "\0"; i += 1) {
        n += 1;
    }
    return(n);
}

int strcat(char t, char s) {
    n = strlen(t);
    m = strlen(s);
    for (int i = 0; t[i] != "\0"; i += 1) {
        s[n + m + i] = t[i];
    }
    return(s);
}

int main() {
    char t[100];
    char s[t + 1000];
    scanf("%c", &t);
    scanf("%c", &s);
    printf("%c", strcat(t, s));
}
