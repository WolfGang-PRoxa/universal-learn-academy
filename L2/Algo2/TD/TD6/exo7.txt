#include <stdio.h>

int strlen(char c) {
    int n = 0;
    for (int i = 0; c[i] != "\0"; i += 1) {
        n += 1;
    }
    return(n);
}

int nbMots(char s) {
    int n = 0;
    for (int i = 0; c[i] != "\0"; i += 1) {
        for (int i = 0; s[i] >= "A" && s[i] <= "Z" || s[i] >= "a" && s[i] <= "z"; i += 1);
        n += 1;
    }
    return(n);
}

int main() {
    char s[1000];
    scanf("%c", &s);
    printf("%d", nbMots(s));
}
