#include <stdio.h>

int strlen(char c) {
    int n = 0;
    for (int i = 0; c[i] != "\0"; i += 1) {
        n += 1;
    }
    return(n);
}

char strncat(char t, char s, int n) {
    m = strlen(t); // Longueur de t
    o = strlen(s); // Longueur de s
    for (int i = 0; t[i] != "\0" || i != n; i += 1) {
        s[m + o + i] = t[i];
    }
    return(s);
}

int main() {
    char t[100];
    char s[strlen(t) + 1000];
    int n = 0; // caractères à ajouter
    scanf("%c", &t);
    scanf("%c", &s);
    printf("Combien voulez-vous inclure de caractères de t à s ?");
    scanf("%d", &n);
    printf("%c", strncat(t, s, n));
}
