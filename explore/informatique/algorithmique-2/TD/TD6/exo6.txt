#include <stdio.h>

int strlen(char c) {
    int n = 0;
    for (int i = 0; c[i] != "\0"; i += 1) {
        n += 1;
    }
    return(n);
}

char atoi(char s) {
    int n = strlen(s);
    char t[n];
    for (int i = 0; s[i] >= "0" && s[i] <= "9" || s[i] == - || s[i] == +; i += 1) {
        t[i] = s[i];
    }
    return(t);
}

int main() {
    char s[1000];
    scanf("%c", &s);
    printf("%c", atoi(s));
}
